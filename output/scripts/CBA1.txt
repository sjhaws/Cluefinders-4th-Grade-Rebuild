  522  set            stopInterruptingScene, kFalse
  523  get_prop       sQueue, actionCount, numActions
  524  loop           (numActions>0)
  525  loop_test      (numActions>0)
  526  jump_if_false  -> 548
  527  if             stopInterruptingScene
  528  jump_if_false  -> 532
  529  set            sceneInterrupted, kFalse
  530  jump           -> 548
  531  end
  532  send           sQueue, halt
  533  get_prop       sQueue, isActive, active
  534  if             active
  535  jump_if_false  -> 539
  536  set            sceneInterrupted, kFalse
  537  jump           -> 548
  538  end
  539  send           sQueue, clear
  540  if             (sQueueFinished#"empty")
  541  jump_if_false  -> 546
  542  set            finishedEvent, sQueueFinished
  543  set            sQueueFinished, "empty"
  544  call           finishedEvent
  545  end
  546  get_prop       sQueue, actionCount, numActions
  547  jump           -> 525
  548  return
