   76  sub            pDisplayObject@76, animate

   80  set            xOff, $result

   93  set            object.objectIndex, RAnimation, ox, oy, objectAOIDs.objectToDisplay
   94  set_prop       object.objectIndex, "z", objectZ

   98  send           object.objectIndex, setAOCursor, kOpenHandCursor, kClosedHandCursor

  100  jump_if_false  -> 112

  137  sub            eObjectDropped@137
  138  get_prop       $1, userData, ud
  139  set            p1, ("#"@ud)
  140  set            p2, (":"@ud)
  141  set            t2, (p1-1)
  142  set            ox, ud|1|t2
  143  set            t1, (p1+1)
  144  set            t2, (p2-p1-1)

  146  set            t1, (p2+1)

  148  set_prop       $1, "z", objectZ
  149  if             (toBackpackOnly=1)
  150  jump_if_false  -> 195
  151  if             (openBackpack#0)
  152  jump_if_false  -> 176
  153  call_global    IntersectTest, "RECEIVER", "openBackpack"
  154  set            test, $result
  155  if             (test=1)
  156  jump_if_false  -> 159
  157  send           openBackpack, addObject, id
  158  jump           -> 175
  159  end
  160  if             (closedBackpack#0)

  163  if             (test=1)
  164  jump_if_false  -> 172
  165  call_global    IntersectTest, "RECEIVER", "closedBackpack"
  166  set            test, $result
  167  if             (test=1)
  168  jump_if_false  -> 171

  171  end
  172  jump           -> 175
  173  end
  174  set            test, 0
  175  jump           -> 194
  176  end
  177  if             (closedBackpack#0)

  179  get_prop       closedBackpack, visible, test

  181  jump_if_false  -> 191
  182  call_global    IntersectTest, "RECEIVER", "closedBackpack"

  189  end
  190  end
  191  jump           -> 194
  192  end
  193  set            test, 0
  194  jump           -> 197
  195  end
  196  set            test, 0
  197  if             (test=1)
  198  jump_if_false  -> 207
  199  get_prop       $1, name, Name

  207  end
  208  set_prop       $1, x, ox
  209  set_prop       $1, y, oy
  210  return
